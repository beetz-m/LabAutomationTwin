\section*{Properties, upper-level placements and the SDT node / referent split}
\paragraph{dul:PhysicalPlace} (stand-in) a place with a physical extent; its sub-places are places (DUL: PhysicalPlace $\sqsubseteq$ PhysicalObject).
\[\begin{array}{l}
\mathsf{dul{:}PhysicalPlace} \sqsubseteq  \mathsf{dul{:}PhysicalObject} \\
\mathsf{dul{:}PhysicalPlace} \sqsubseteq  \exists \mathit{hasSubPlace}.\mathsf{dul{:}PhysicalPlace} \\
\end{array}\]
\paragraph{LabArtifact} A designed physical object of the lab that the twin represents.
\[\begin{array}{l}
\mathsf{LabArtifact} \sqsubseteq  \mathsf{dul{:}PhysicalObject} \\
\mathsf{LabArtifact} \sqsubseteq  \exists \mathit{isRepresentedBy}.\mathsf{LabSDTNode} \\
\end{array}\]
\paragraph{LabSDTNode} A node of the lab's semantic digital twin (one USD prim): information that represents one entity.
\[\begin{array}{l}
\mathsf{LabSDTNode} \sqsubseteq  \mathsf{SceneNode} \\
\mathsf{LabSDTNode} \sqsubseteq  \exists \mathit{representsEntity}.\mathsf{Entity} \\
\mathsf{LabSDTNode} \sqsubseteq  \exists \mathit{usdPrimPath}.\mathsf{xsd{:}string} \\
\end{array}\]
\paragraph{GroupingNode} A structural SDT node (USD scope / xform) that groups child nodes; it represents the grouping only.
\[\begin{array}{l}
\mathsf{GroupingNode} \sqsubseteq  \mathsf{LabSDTNode} \\
\mathsf{GroupingNode} \sqsubseteq  \exists \mathit{hasChildNode}.\mathsf{LabSDTNode} \\
\end{array}\]
\paragraph{LabSemanticDigitalTwin} The semantic digital twin of a laboratory site: the belief state about it, structured as a USD scene graph.
\[\begin{array}{l}
\mathsf{LabSemanticDigitalTwin} \sqsubseteq  \mathsf{SemanticDigitalTwin} \\
\mathsf{LabSemanticDigitalTwin} \sqsubseteq  \exists \mathit{hasNodeStructure}.\mathsf{LabSceneHierarchy} \\
\end{array}\]
\paragraph{LabSceneHierarchy} The USD namespace hierarchy of the lab twin (lab\_twin.usda + culture layer).
\[\begin{array}{l}
\mathsf{LabSceneHierarchy} \sqsubseteq  \mathsf{HierarchicalSceneStructure} \\
\mathsf{LabSceneHierarchy} \sqsubseteq  \exists \mathit{hasNode}.\mathsf{LabSDTNode} \\
\end{array}\]
\paragraph{SimulationSettings} Physics or rendering settings of the twin (physics scene, sky and sun lights).
\[\begin{array}{l}
\mathsf{SimulationSettings} \sqsubseteq  \mathsf{dul{:}Description} \\
\mathsf{SimulationSettings} \sqsubseteq  \exists \mathit{configures}.\mathsf{LabSemanticDigitalTwin} \\
\end{array}\]
\paragraph{SimulationSettingsNode} An SDT node holding settings.
\[\begin{array}{l}
\mathsf{SimulationSettingsNode} \sqsubseteq  \mathsf{LabSDTNode} \\
\mathsf{SimulationSettingsNode} \sqsubseteq  \exists \mathit{representsEntity}.\mathsf{SimulationSettings} \\
\end{array}\]
\paragraph{RenderRepresentationNode} A render-only USD prim (looks, materials, collision / visual geometry): representation, not SDT knowledge.
\[\begin{array}{l}
\mathsf{RenderRepresentationNode} \sqsubseteq  \mathsf{SceneNode} \\
\mathsf{RenderRepresentationNode} \sqsubseteq  \neg \mathsf{LabSDTNode} \\
\end{array}\]
