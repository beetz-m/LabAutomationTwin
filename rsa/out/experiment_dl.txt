EXPERIMENTS
  -- Experiment: an activity that answers a question by producing insights
  Experiment ⊑ Process
  Experiment ⊑ (=1 hasQuestion.Question)
  Experiment ⊑ (=1 hasGoal.⊤)
  Experiment ⊑ ∃hasInsight.Insight
     further constraints: parthood-preserves-kind
  -- HypothesisDiscriminatingExperiment: the question asks which of several candidate hypotheses holds; the experiment changes a belief
over them, and its answer is the posterior
  HypothesisDiscriminatingExperiment ⊑ Experiment
  HypothesisDiscriminatingExperiment ⊑ ∃hasQuestion.(≥2 hasCandidate.⊤)
  HypothesisDiscriminatingExperiment ⊑ (=1 hasBelief.(Belief ⊓ ∃over.⊤ ⊓ ∃hasPrior.⊤))
  HypothesisDiscriminatingExperiment ⊑ (=1 hasGoal.(Belief ⊓ ∃expectedInformationGain.⊤))
  HypothesisDiscriminatingExperiment ⊑ (=1 hasInsight.(Insight ⊓ ∃hasRole.AnswerRole ⊓ ∃hasForm.Posterior))
     further constraints: answer-from-a-comparison-that-updates-the-belief
  -- ComparativeExperiment: units are treated differently, kept under shared conditions and observed the same way;
comparisons between units determine the effects of the treatments. Its procedure yields three phases
  ComparativeExperiment ⊑ Experiment
  ComparativeExperiment ⊑ (≥2 hasUnit.∃hasTreatment.⊤)
  ComparativeExperiment ⊑ (=1 hasSharedConditions.(Condition ⊓ ∃fromTime.⊤ ⊓ ∃toTime.⊤ ⊓ ∃hasObservation.(Observation ⊓ ∃atTime.⊤ ⊓ ∃interpretedAs.⊤)))
  ComparativeExperiment ⊑ ∃hasComparison.(Comparison ⊓ ∃between.⊤ ⊓ ∃determines.Effect)
  ComparativeExperiment ⊑ ∃hasInsight.(Insight ⊓ ∃hasRole.EffectRole)
  ComparativeExperiment ⊑ (=1 hasPhase.CultivationPhase)
  ComparativeExperiment ⊑ (=1 hasPhase.ObservationPhase)
  ComparativeExperiment ⊑ (=1 hasPhase.AnalysisPhase)
  ComparativeExperiment ⊑ (=1 hasSite.Plate)
     further constraints: comparisons-between-units, an-effect-insight-per-comparison, parts-conform-to-their-concepts, each-step-in-one-phase, each-exposure-realized-once, each-unit-observed-once, each-comparison-realized-once, units-in-distinct-wells-of-the-site, phases-in-time-order
  -- ControlledExperiment: one unit is an untreated control, it stays untreated, and every comparison isolates exactly one
exposure: its two units differ in that exposure only, and it is the exposure whose effect the comparison determines
  ControlledExperiment ⊑ ComparativeExperiment
  ControlledExperiment ⊑ (=1 hasUnit.(∃hasRole.ControlRole ⊓ ∃hasTreatment.EmptyTreatment))
  ControlledExperiment ⊑ ∃prevents.(Exposure ⊓ ∃of.(Culture ⊓ ∃hasRole.ControlRole))
     further constraints: single-difference
  -- ChallengeRescueExperiment: units are injured by a challenge; one is then given a rescue after a delay. A validity comparison
(comparator vs control) establishes the challenge's effect and licenses the primary comparison (rescued vs
comparator), which determines the rescue's effect and updates the belief over models of recovery
  ChallengeRescueExperiment ⊑ ControlledExperiment ⊓ HypothesisDiscriminatingExperiment
  ChallengeRescueExperiment ⊑ (=1 hasSubstance.(Substance ⊓ ∃hasRole.ChallengeRole))
  ChallengeRescueExperiment ⊑ (=1 hasSubstance.(Substance ⊓ ∃hasRole.RescueRole))
  ChallengeRescueExperiment ⊑ (=1 hasUnit.(Culture ⊓ ∃hasRole.RescuedRole ⊓ ∃hasTreatment.(Treatment ⊓ ∃hasExposure.(Exposure ⊓ ∃exposedTo.(Substance ⊓ ∃hasRole.ChallengeRole)) ⊓ ∃hasExposure.(Exposure ⊓ ∃exposedTo.(Substance ⊓ ∃hasRole.RescueRole)))))
  ChallengeRescueExperiment ⊑ (=1 hasUnit.(Culture ⊓ ∃hasRole.ComparatorRole ⊓ ∃hasTreatment.(Treatment ⊓ ∃hasExposure.(Exposure ⊓ ∃exposedTo.(Substance ⊓ ∃hasRole.ChallengeRole)))))
  ChallengeRescueExperiment ⊑ (=1 hasComparison.(Comparison ⊓ ∃hasRole.ValidityRole ⊓ ∃between.(Culture ⊓ ∃hasRole.ComparatorRole) ⊓ ∃between.(Culture ⊓ ∃hasRole.ControlRole) ⊓ ∃determines.(Effect ⊓ ∃of.(Substance ⊓ ∃hasRole.ChallengeRole)) ⊓ ∃licenses.(Comparison ⊓ ∃hasRole.PrimaryRole)))
  ChallengeRescueExperiment ⊑ (=1 hasComparison.(Comparison ⊓ ∃hasRole.PrimaryRole ⊓ ∃between.(Culture ⊓ ∃hasRole.RescuedRole) ⊓ ∃between.(Culture ⊓ ∃hasRole.ComparatorRole) ⊓ ∃determines.(Effect ⊓ ∃of.(Substance ⊓ ∃hasRole.RescueRole)) ⊓ ∃updates.Belief))
     further constraints: rescue-after-challenge

CONSTRAINTS BEYOND THE TBOX AXIOMS
  parthood-preserves-kind [OWL 2 DL]
      Process ⊑ ∀hasPart.Process
      MaterialEntity ⊑ ∀hasPart.MaterialEntity
      PlanSpecification ⊑ ∀hasPart.PlanSpecification
      InformationEntity ⊑ ∀hasPart.InformationEntity
      (stated in the kinds section; parthood never crosses kinds)
  answer-from-a-comparison-that-updates-the-belief [OWL 2 DL]
      Insight ⊓ ∃hasRole.AnswerRole ⊑ ∃from.(Comparison ⊓ ∃updates.Belief)
      (that the comparison is one of the same experiment needs variable sharing (FOL))
  comparisons-between-units [SROIQ role inclusion]
      hasComparison ∘ between ⊑ hasUnit
      (read as an integrity constraint: under the open world an OWL reasoner would infer the unit, not report it)
  an-effect-insight-per-comparison [FOL]
      ∀e,c. hasComparison(e,c) → ∃i. hasInsight(e,i) ∧ from(i,c) ∧ ∀s. (determines∘of)(c,s) → of(i,s)
      (OWL 2 states only Comparison ⊑ ∃from⁻.(Insight ⊓ ∃hasRole.EffectRole))
  parts-conform-to-their-concepts [DL semantics]
      (every TBox axiom holds of every individual)
      (the closed-world checker evaluates each part against its own concept)
  each-step-in-one-phase [OWL 2 DL]
      Action ⊑ (=1 hasStep⁻.Phase)
  each-exposure-realized-once [OWL 2 DL + rule]
      Exposure ⊑ (=1 realizes⁻.InterventionAction)
      realizes(a,x) ← hasCulture(a,u) ∧ hasTreatment(u,t) ∧ hasExposure(t,x) ∧  same substance, dose and time
      (the rule that computes realizes compares data values of two individuals (SWRL))
  each-unit-observed-once [FOL]
      ∀e,u. hasUnit(e,u) → ∃=1 s. hasPart(e,s) ∧ ObservationAction(s) ∧ hasCulture(s,u)
      (OWL 2 states only Culture ⊑ (=1 hasCulture⁻.ObservationAction), not 'within the same experiment')
  each-comparison-realized-once [OWL 2 DL + rule]
      Comparison ⊑ (=1 realizes⁻.AnalysisAction)
      (realizes is computed from the analysis's statistic (the pair of units it compares))
  units-in-distinct-wells-of-the-site [FOL (path agreement)]
      Culture ⊑ (=1 containedIn.Well),   Well ⊑ (≤1 contains.⊤)
      ∀e,u,v. hasUnit(e,u) ∧ hasUnit(e,v) → (containedIn∘partOf)(u) = (containedIn∘partOf)(v)
      (the first line is OWL 2 DL; 'one plate for all units' is a role-value map, undecidable with the rest)
  phases-in-time-order [ALC(D) feature chains]
      ComparativeExperiment ⊑ ∃(cultivation∘toTime, observationPhase∘atTime).≤
      (compares two data values reached by different paths: a concrete-domain predicate, not available in OWL 2)
  single-difference [FOL with counting]
      ∀c,u,v. between(c,u) ∧ between(c,v) ∧ u≠v → |X(u) △ X(v)| = 1 ∧ ∀x ∈ X(u) △ X(v). exposedTo(x) = (determines∘of)(c)
      (X(u): the exposures of the treatment of u, as substance, dose and time)
      (beyond SROIQ: compares two individuals' sets of exposures (rule with aggregation, SPARQL, or the checker))
  rescue-after-challenge [ALC(D) feature chains]
      ∃hasRole.RescuedRole ⊑ ∃(rescueExposure∘atTime, challengeExposure∘atTime).>
      (with rescueExposure, challengeExposure functional subroles of hasTreatment∘hasExposure; not OWL 2)
